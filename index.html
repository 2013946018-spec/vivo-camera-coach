0 · 连接公开相册服务

网页代码还没有填入 Supabase 项目配置。完成下方设置后，公开相册和上传功能才会启用；摄像头本地指导仍可先使用。

填入三个值后，网页会在手机上生成配置好的 index.html 文件，无需手动改代码。
配置步骤与安全说明
  1. 创建一个 Supabase 免费项目。
  2. 在 Supabase Authentication 设置里暂时关闭 Email Confirmations。托管项目的默认邮件服务可能无法向普通邮箱发送验证邮件。
  3. 在 Project Settings → API Keys 页面复制 Project URL 和 Publishable key（旧项目可能显示 anon public key）。
  4. 把 URL、公钥和相册所有者邮箱填入上方三个框，点击“生成配置版 index.html”，保存下载的文件。
  5. 展开下方 SQL，点击“复制 SQL”（会自动填入你刚刚输入的邮箱），粘贴到 Supabase 的 SQL Editor 并执行。
  6. 把配置版文件上传到新的 GitHub 公共仓库，文件名设为 index.html，再开启 GitHub Pages。
  7. 打开公开网页，用指定邮箱注册并登录。成功后立刻回 Supabase Authentication 设置，关闭 Allow new users to sign up。数据库规则也会把上传/删除权限限制为你填写的邮箱。

只能使用 Publishable key / anon public key。绝对不要把 service_role 或 Secret key 填进网页，它们拥有绕过权限规则的能力。

复制 Supabase SQL（首次设置时执行一次）
create table if not exists public.selfies (
  id uuid primary key default gen_random_uuid(),
  path text not null unique,
  caption text not null default '',
  created_at timestamptz not null default now(),
  owner_id uuid not null references auth.users(id) on delete cascade
);

alter table public.selfies enable row level security;
grant select on public.selfies to anon, authenticated;
grant insert, delete on public.selfies to authenticated;

drop policy if exists "Public can view gallery" on public.selfies;
create policy "Public can view gallery" on public.selfies
for select to anon, authenticated using (true);

drop policy if exists "Signed-in owner can add gallery rows" on public.selfies;
create policy "Signed-in owner can add gallery rows" on public.selfies
for insert to authenticated with check (auth.uid() = owner_id and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_LOGIN_EMAIL@example.com'));

drop policy if exists "Owner can delete own gallery rows" on public.selfies;
create policy "Owner can delete own gallery rows" on public.selfies
for delete to authenticated using (auth.uid() = owner_id and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_LOGIN_EMAIL@example.com'));

insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values ('public-selfies', 'public-selfies', true, 10485760,
        array['image/jpeg','image/png','image/webp'])
on conflict (id) do update set
  public = excluded.public,
  file_size_limit = excluded.file_size_limit,
  allowed_mime_types = excluded.allowed_mime_types;

drop policy if exists "Owner can upload selfie files" on storage.objects;
create policy "Owner can upload selfie files" on storage.objects
for insert to authenticated
with check (
  bucket_id = 'public-selfies'
  and (storage.foldername(name))[1] = auth.uid()::text
  and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_LOGIN_EMAIL@example.com')
);

drop policy if exists "Owner can delete selfie files" on storage.objects;
create policy "Owner can delete selfie files" on storage.objects
for delete to authenticated
using (
  bucket_id = 'public-selfies'
  and (storage.foldername(name))[1] = auth.uid()::text
  and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_LOGIN_EMAIL@example.com')
);
